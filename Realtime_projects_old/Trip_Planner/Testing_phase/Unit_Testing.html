Unit Testing & Tool Validation

Unit tests focus on the individual components of our system: the MCP tool wrappers, the FastAPI routers, and the react utility components.

Backend Unit Tests (Pytest)

import pytest
from app.tools import weather_tool

@pytest.mark.asyncio
async def test_weather_tool_valid_coords():
    # Test with Paris coordinates
    result = await weather_tool.invoke({"lat": 48.8566, "lon": 2.3522})
    assert "temperature" in result
    assert result["status"] == "success"

def test_pydantic_schema_validation():
    # Ensure tool inputs are strictly validated
    with pytest.raises(ValidationError):
        weather_tool.invoke({"lat": "invalid_string"})
